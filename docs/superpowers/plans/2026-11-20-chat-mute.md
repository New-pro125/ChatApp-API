# Chat Mute Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a caller-private per-chat mute setting with `8h`, `7d`, and forever presets, lazy expiry, a single `PATCH` API, and chat-view projection.

**Architecture:** Store `muted_at`/`muted_until` on `chat_members`, compute active state lazily through a `MuteState` domain class, and expose it through a focused `MuteService` and request-scoped `MuteController`. No notification delivery, unread-count, or mention behavior changes.

**Tech Stack:** Node 24, TypeScript 7, Express 5, Inversify 8, Drizzle ORM, PostgreSQL 18, Zod, OpenAPI 3.1, Vitest/Supertest, Biome.

**Spec:** `docs/superpowers/specs/2026-11-20-chat-mute-design.md`

## Global Constraints

- Setting/chat-view only. No push, in-app notification, or notification-dispatch abstraction is added in this phase.
- Muting does not change message delivery, unread counts, read receipts, mention rows, or chat list ordering.
- Presets only: `8h`, `7d`, `forever`, or `null` to unmute. Clients cannot send arbitrary durations or timestamps.
- All existing chats are supported: direct and secret now; group/channel later inherit the same columns.
- Single `PATCH /api/chats/:chatId/mute` endpoint; `{ "preset": null }` unmutes.
- No new error codes, configuration values, auth schemes, or `.env.example` changes.
- Expired rows are reported as unmuted but are not rewritten during reads. Setting or clearing overwrites stale values.
- No background cleanup job and no new config.
- Future Phase 5 notification code must reuse `MuteState.from`; it must not reimplement the predicate.
- `plan.md` stays untracked/gitignored. Never commit `.env`, secrets, tokens, or credentials.
- Do not run multiple Vitest processes concurrently; they share the test database and `clearDatabase()` causes false 401 failures.

## Review Focus

- Expired timed mute must be reported as unmuted with `muteExpiresAt: null` and must not be rewritten on read — pinned by the expired-row test in Task 6.
- Forever must remain distinct from unmuted (`mutedAt` set, `mutedUntil` null) — pinned by `MuteState.from` in Task 1 and the PATCH round-trip in Task 5.
- A peer's mute must never be visible in the caller's chat views — pinned by the caller-private view test in Task 6.
- A non-member must not be able to set a mute, including on someone else's chat id — pinned by the membership test in Task 5.
- Chat-list hydration through `listByChats` must carry mute fields without N+1 writes — pinned by the list-view assertion in Task 6.

```ts
export const MUTE_PRESETS = ['8h', '7d', 'forever'] as const;

export type MutePresetValue = (typeof MUTE_PRESETS)[number];

export interface MuteExpiry {
  readonly mutedAt: Date;
  readonly mutedUntil: Date | null;
}

const DURATION_MS: Record<Exclude<MutePresetValue, 'forever'>, number> = {
  '8h': 8 * 60 * 60 * 1000,
  '7d': 7 * 24 * 60 * 60 * 1000,
};

export class MutePreset {
  static readonly values = MUTE_PRESETS;

  private constructor(private readonly value: MutePresetValue) {}

  static resolve(preset: MutePresetValue, now: Date): MuteExpiry {
    return new MutePreset(preset).resolve(now);
  }

  private resolve(now: Date): MuteExpiry {
    if (this.value === 'forever') {
      return { mutedAt: now, mutedUntil: null };
    }

    return { mutedAt: now, mutedUntil: new Date(now.getTime() + DURATION_MS[this.value]) };
  }
}
```

```ts
export interface MuteMembership {
  readonly mutedAt: Date | null;
  readonly mutedUntil: Date | null;
}

export class MuteState {
  private constructor(
    readonly muted: boolean,
    readonly expiresAt: Date | null,
  ) {}

  static from(membership: MuteMembership, now: Date): MuteState {
    if (membership.mutedAt === null) {
      return new MuteState(false, null);
    }

    if (membership.mutedUntil === null) {
      return new MuteState(true, null);
    }

    if (membership.mutedUntil.getTime() <= now.getTime()) {
      return new MuteState(false, null);
    }

    return new MuteState(true, membership.mutedUntil);
  }
}
```

```ts
  async setMute(
    chatId: string,
    userId: string,
    mutedAt: Date | null,
    mutedUntil: Date | null,
  ): Promise<ChatMemberRecord> {
    try {
      const rows = await this.db
        .update(chatMembers)
        .set({ mutedAt, mutedUntil })
        .where(and(eq(chatMembers.chatId, chatId), eq(chatMembers.userId, userId)))
        .returning(chatMemberSelection);

      const updated = rows[0];

      if (!updated) {
        throw new PersistenceException('Failed to set chat mute: membership not found');
      }

      return this.mapper.toRecord(updated);
    } catch (error) {
      if (error instanceof PersistenceException) {
        throw error;
      }

      throw new PersistenceException('Failed to set chat mute', { cause: error });
    }
  }
```

```ts
import { inject, injectable } from 'inversify';
import {
  MutePreset,
  type MutePresetValue,
} from '@/modules/messages/logic/entities/mute-preset.js';
import { MuteState } from '@/modules/messages/logic/entities/mute-state.js';
import type { MessagingPolicy } from '@/modules/messages/logic/messaging-policy.service.js';
import type { ChatMemberRepository } from '@/modules/messages/logic/ports/chat-member.repository.js';
import { MESSAGES_TYPES } from '@/modules/messages/messages.types.js';

export interface MuteSettingResult {
  readonly chatId: string;
  readonly muted: boolean;
  readonly muteExpiresAt: Date | null;
}

/** Applies caller-scoped mute presets after checking chat membership. */
@injectable()
export class MuteService {
  constructor(
    @inject(MESSAGES_TYPES.MessagingPolicy) private readonly policy: MessagingPolicy,
    @inject(MESSAGES_TYPES.ChatMemberRepository) private readonly members: ChatMemberRepository,
  ) {}

  async setMute(
    userId: string,
    chatId: string,
    preset: MutePresetValue | null,
    now: Date = new Date(),
  ): Promise<MuteSettingResult> {
    await this.policy.requireMember(chatId, userId);

    const expiry =
      preset === null ? { mutedAt: null, mutedUntil: null } : MutePreset.resolve(preset, now);
    const member = await this.members.setMute(chatId, userId, expiry.mutedAt, expiry.mutedUntil);
    const state = MuteState.from(member, now);
```

```ts
import type { Request, RequestHandler } from 'express';
import { inject, injectable } from 'inversify';
import { CORE_TYPES } from '@/core/container/core.types.js';
import { UnauthorizedException } from '@/core/errors/http.error.js';
import type { ApiResponse } from '@/core/http/response/api-response.js';
import type { MuteService } from '@/modules/messages/logic/mute.service.js';
import { MESSAGES_TYPES } from '@/modules/messages/messages.types.js';
import {
  chatParamsSchema,
  setMuteSchema,
} from '@/modules/messages/web/dto/message.schemas.js';

/** Request-scoped handlers for the caller's per-chat mute setting. */
@injectable()
export class MuteController {
  constructor(
    @inject(MESSAGES_TYPES.MuteService) private readonly mutes: MuteService,
    @inject(CORE_TYPES.ApiResponse) private readonly response: ApiResponse,
  ) {}

  set: RequestHandler = async (req, res) => {
    const user = this.requireUser(req);
    const { chatId } = chatParamsSchema.parse(req.params);
    const { preset } = setMuteSchema.parse(req.body);
    const result = await this.mutes.setMute(user.id, chatId, preset);

    this.response.ok(res, result);
  };

  private requireUser(req: Request): Express.User {
    if (!req.isAuthenticated()) {
      throw new UnauthorizedException();
    }

    return req.user;
  }
}
```
